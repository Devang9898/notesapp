Schema::create('notes', function (Blueprint $table) {
    $table->id();
    $table->foreignId('user_id')->constrained()->onDelete('cascade');
    $table->string('title');
    $table->text('content');
    $table->string('tags')->nullable();
    $table->boolean('is_favorite')->default(false);
    $table->enum('status', ['active', 'archived'])->default('active');
    $table->string('color')->nullable();
    $table->timestamp('reminder_at')->nullable();
    $table->text('attachments')->nullable();
    $table->softDeletes();
    $table->timestamps();
});

### ✅ Controller Logic (Add to `NoteController`)
```php
public function bookmarks(Request $request) {
    $notes = Note::where('user_id', Auth::id())
                ->where('is_favorite', true)
                ->where('status', '!=', 'archived')
                ->whereNull('deleted_at')
                ->get();
    return view('notes.bookmarks', compact('notes'));
}

public function archives(Request $request) {
    $notes = Note::where('user_id', Auth::id())
                ->where('status', 'archived')
                ->where('is_favorite', false)
                ->whereNull('deleted_at')
                ->get();
    return view('notes.archives', compact('notes'));
}

public function trash() {
    $notes = Note::onlyTrashed()->where('user_id', Auth::id())->get();
    return view('notes.trash', compact('notes'));
}

public function restore($id) {
    $note = Note::onlyTrashed()->findOrFail($id);
    $note->restore();
    return redirect()->route('notes.trash')->with('success', 'Note restored.');
}

public function forceDelete($id) {
    $note = Note::onlyTrashed()->findOrFail($id);
    $note->forceDelete();
    return redirect()->route('notes.trash')->with('success', 'Note permanently deleted.');
}

public function bulkAction(Request $request) {
    $request->validate([
        'action' => 'required|in:delete,restore,archive',
        'note_ids' => 'required|array'
    ]);

    $ids = $request->note_ids;

    switch ($request->action) {
        case 'delete':
            Note::whereIn('id', $ids)->delete();
            break;
        case 'restore':
            Note::onlyTrashed()->whereIn('id', $ids)->restore();
            break;
        case 'archive':
            Note::whereIn('id', $ids)->update(['status' => 'archived']);
            break;
    }

    return back()->with('success', 'Bulk action executed successfully.');
}
```

<?php

namespace App\Http\Controllers;

use App\Models\Note;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class NoteController extends Controller
{
    public function index(Request $request)
    {
        $query = Note::where('user_id', Auth::id())->whereNull('deleted_at');

        if ($request->filled('search')) {
            $search = $request->search;
            $query->where(function ($q) use ($search) {
                $q->where('title', 'like', "%{$search}%")
                  ->orWhere('content', 'like', "%{$search}%");
            });
        }

        if ($request->sort === 'name') {
            $query->orderBy('title');
        } elseif ($request->sort === 'date_asc') {
            $query->orderBy('created_at', 'asc');
        } else {
            $query->orderBy('created_at', 'desc');
        }

        $notes = $query->get();

        return view('notes.index', compact('notes'));
    }

    public function create()
    {
        return view('notes.create');
    }

    public function store(Request $request)
    {
        $request->validate([
            'title' => 'required|string|max:255',
            'content' => 'required',
            'tags' => 'nullable|string',
            'status' => 'required|in:active,archived',
            'reminder_at' => 'nullable|date',
            'attachments' => 'nullable|string',
        ]);

        Note::create([
            'user_id' => Auth::id(),
            'title' => $request->title,
            'content' => $request->content,
            'tags' => $request->tags,
            'status' => $request->status,
            'reminder_at' => $request->reminder_at,
            'attachments' => $request->attachments,
            'is_favorite' => $request->has('is_favorite'),
        ]);

        return redirect()->route('notes.index')->with('success', 'Note created successfully.');
    }

    public function show(Note $note)
    {
        $this->authorizeNote($note);
        return view('notes.show', compact('note'));
    }

    public function edit(Note $note)
    {
        $this->authorizeNote($note);
        return view('notes.edit', compact('note'));
    }

    public function update(Request $request, Note $note)
    {
        $this->authorizeNote($note);

        $request->validate([
            'title' => 'required|string|max:255',
            'content' => 'required',
            'tags' => 'nullable|string',
            'status' => 'required|in:active,archived',
            'reminder_at' => 'nullable|date',
            'attachments' => 'nullable|string',
        ]);

        $note->update([
            'title' => $request->title,
            'content' => $request->content,
            'tags' => $request->tags,
            'status' => $request->status,
            'reminder_at' => $request->reminder_at,
            'attachments' => $request->attachments,
            'is_favorite' => $request->has('is_favorite'),
        ]);

        return redirect()->route('notes.index')->with('success', 'Note updated successfully.');
    }

    public function destroy(Note $note)
    {
        $this->authorizeNote($note);
        $note->delete();

        return redirect()->route('notes.index')->with('success', 'Note moved to trash.');
    }

    public function bookmarks()
    {
        $notes = Note::where('user_id', Auth::id())
            ->where('is_favorite', true)
            ->whereNull('deleted_at')
            ->get();

        return view('notes.bookmarks', compact('notes'));
    }

    public function archives()
    {
        $notes = Note::where('user_id', Auth::id())
            ->where('status', 'archived')
            ->whereNull('deleted_at')
            ->get();

        return view('notes.archives', compact('notes'));
    }

    public function trash()
    {
        $notes = Note::onlyTrashed()->where('user_id', Auth::id())->get();
        return view('notes.trash', compact('notes'));
    }

    public function restore($id)
    {
        $note = Note::onlyTrashed()->where('user_id', Auth::id())->findOrFail($id);
        $note->restore();

        return redirect()->route('notes.trash')->with('success', 'Note restored successfully.');
    }

    public function forceDelete($id)
    {
        $note = Note::onlyTrashed()->where('user_id', Auth::id())->findOrFail($id);
        $note->forceDelete();

        return redirect()->route('notes.trash')->with('success', 'Note permanently deleted.');
    }

    public function toggleFavorite(Note $note)
    {
        $this->authorizeNote($note);
        $note->is_favorite = !$note->is_favorite;
        $note->save();

        return redirect()->back()->with('success', 'Bookmark updated.');
    }

    public function bulkAction(Request $request)
    {
        $request->validate([
            'note_ids' => 'required|array',
            'action' => 'required|string|in:delete,restore,force-delete',
        ]);

        $notes = Note::withTrashed()->whereIn('id', $request->note_ids)->where('user_id', Auth::id());

        switch ($request->action) {
            case 'delete':
                $notes->delete();
                return back()->with('success', 'Selected notes moved to trash.');
            case 'restore':
                $notes->restore();
                return back()->with('success', 'Selected notes restored.');
            case 'force-delete':
                $notes->forceDelete();
                return back()->with('success', 'Selected notes permanently deleted.');
        }

        return back()->with('error', 'Invalid action.');
    }

    private function authorizeNote(Note $note)
    {
        if ($note->user_id !== Auth::id()) {
            abort(403);
        }
    }
}

        return view('notes.bookmarks', compact('notes'));
    }

    public function archives()
    {
        $notes = Note::where('user_id', Auth::id())
            ->where('status', 'archived')
            ->whereNull('deleted_at')
            ->get();

        return view('notes.archives', compact('notes'));
    }

    public function trash()
    {
        $notes = Note::onlyTrashed()->where('user_id', Auth::id())->get();
        return view('notes.trash', compact('notes'));
    }

    public function restore($id)
    {
        $note = Note::onlyTrashed()->where('user_id', Auth::id())->findOrFail($id);
        $note->restore();

        return redirect()->route('notes.trash')->with('success', 'Note restored successfully.');
    }

    public function forceDelete($id)
    {
        $note = Note::onlyTrashed()->where('user_id', Auth::id())->findOrFail($id);
        $note->forceDelete();

        return redirect()->route('notes.trash')->with('success', 'Note permanently deleted.');
    }

    public function toggleFavorite(Note $note)
    {
        $this->authorizeNote($note);
        $note->is_favorite = !$note->is_favorite;
        $note->save();

        return redirect()->back()->with('success', 'Bookmark updated.');
    }

    public function bulkAction(Request $request)
    {
        $request->validate([
            'note_ids' => 'required|array',
            'action' => 'required|string|in:delete,restore,force-delete',
        ]);

        $notes = Note::withTrashed()->whereIn('id', $request->note_ids)->where('user_id', Auth::id());

        switch ($request->action) {
            case 'delete':
                $notes->delete();
                return back()->with('success', 'Selected notes moved to trash.');
            case 'restore':
                $notes->restore();
                return back()->with('success', 'Selected notes restored.');
            case 'force-delete':
                $notes->forceDelete();
                return back()->with('success', 'Selected notes permanently deleted.');
        }

        return back()->with('error', 'Invalid action.');
    }

    private function authorizeNote(Note $note)
    {
        if ($note->user_id !== Auth::id()) {
            abort(403);
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Http\Requests\RegisterRequest;
use App\Http\Requests\LoginRequest;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Str;
use App\Models\User;
use App\Mail\OtpMail; // We'll create this Mailable next
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

class WebAuthController extends Controller
{
    // Show the login form
    public function showLoginForm()
    {
        return view('auth.login');
    }

    // Handle login process
    public function login(LoginRequest $request): RedirectResponse
    {
        // First, attempt to authenticate the user with the credentials
        $credentials = $request->only('email', 'password');

        if (Auth::attempt($credentials)) {
            // Generate OTP and store it in the session
            $otp = Str::random(6); // 6 character OTP
            session(['otp' => $otp, 'otp_sent_to' => Auth::user()->email]);

            // Send OTP to user's email
            Mail::to(Auth::user()->email)->send(new OtpMail($otp));

            // Redirect to OTP verification page
            return redirect()->route('verify-otp');
        }

        return redirect()->back()->withErrors(['email' => 'Invalid credentials']);
    }

    // Show OTP verification form
    public function showOtpForm()
    {
        return view('auth.verify-otp');
    }

    // Handle OTP verification
    public function verifyOtp(Request $request)
    {
        // Validate OTP input
        $request->validate([
            'otp' => 'required|numeric|digits:6',
        ]);

        // Check if the entered OTP matches the stored OTP
        if ($request->otp == session('otp')) {
            // OTP is correct, log the user in
            session()->forget('otp'); // Remove OTP from session

            return redirect()->route('notes.index');
        }

        return redirect()->back()->withErrors(['otp' => 'Invalid OTP']);
    }

    // Logout user
    public function logout(): RedirectResponse
    {
        Auth::logout();
        return redirect('/')->with('status', 'You have been logged out.');
    }
}

    // Handle User Login (First Step)
    public function login(LoginRequest $request)
    {
        $credentials = $request->only('email', 'password');

        // Attempt login with provided credentials
        if (Auth::attempt($credentials)) {
            // Get the registered user's email from the database
            $user = Auth::user();

            // Generate OTP
            $otp = rand(100000, 999999);

            // Store OTP in session
            session(['otp' => $otp, 'otp_sent_to' => $user->email]);

            // Send OTP to the user's registered email

            // Redirect to OTP verification page

        // Validate OTP input
        $request->validate([
            'otp' => 'required|numeric',
        ]);

        // Check if OTP matches

<script>
    document.addEventListener('DOMContentLoaded', () => {
        const searchInput = document.getElementById('noteSearch');
        const resultsList = document.getElementById('autocomplete-results');

        searchInput.addEventListener('input', function () {
            const query = this.value.trim();

            if (query.length < 2) {
                resultsList.classList.add('hidden');
                resultsList.innerHTML = '';
                return;
            }

            fetch(`{{ route('notes.search') }}?query=${encodeURIComponent(query) }`)
                .then(res => res.json())
                .then(data => {
                    resultsList.innerHTML = '';
                    if (data.length === 0) {
                        resultsList.innerHTML = '<li class="px-4 py-2 text-gray-500">No matching notes found</li>';
                    } else {
                        data.forEach(note => {
                            const li = document.createElement('li');
                            li.textContent = note.title;
                            li.className = 'px-4 py-2 hover:bg-blue-100 cursor-pointer';
                            li.onclick = () => window.location.href = `/notes/${note.id}`;
                            resultsList.appendChild(li);
                        });
                    }

                    resultsList.classList.remove('hidden');
                })
                .catch(err => {
                    console.error('Live search failed:', err);
                    resultsList.classList.add('hidden');
                });
        });

        document.addEventListener('click', function (e) {
            if (!searchInput.contains(e.target) && !resultsList.contains(e.target)) {
                resultsList.classList.add('hidden');
            }
        });
    });
</script>
